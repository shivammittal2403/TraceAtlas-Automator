def read_audit() -> dict[str, Any]:
    if AUDIT_PATH.exists():
        with AUDIT_PATH.open('r', encoding='utf-8') as f:
            return json.load(f)
    return json.loads(zlib.decompress(base64.b85decode(_EMBEDDED_AUDIT_B85)).decode('utf-8'))


def import_core():
    if globals().get('_TRACEATLAS_ALL_IN_ONE'):
        return sys.modules[__name__]
    if not CORE_PATH.exists():
        raise FileNotFoundError('Missing 19-domain baseline: ' + str(CORE_PATH))
    name = 'traceatlas_unified_core'
    if name in sys.modules:
        return sys.modules[name]
    spec = importlib.util.spec_from_file_location(name, CORE_PATH)
    if not spec or not spec.loader:
        raise ImportError('Cannot import TraceAtlas unified baseline')
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    try:
        spec.loader.exec_module(module)
    except BaseException:
        sys.modules.pop(name, None)
        raise
    return module

class StudioHandler(http.server.BaseHTTPRequestHandler):

    def log_message(self, fmt, *args):
        log.info('%s %s', self.address_string(), fmt % args)

    def _reply(self, code: int, raw: bytes, kind: str = 'application/json; charset=utf-8'):
        self.send_response(code)
        self.send_header('Content-Type', kind)
        self.send_header('X-Content-Type-Options', 'nosniff')
        self.send_header('X-Frame-Options', 'DENY')
        self.send_header('Referrer-Policy', 'no-referrer')
        self.send_header('Cache-Control', 'no-store')
        self.send_header('Content-Security-Policy', "default-src 'self'; script-src 'unsafe-inline'; style-src 'unsafe-inline'; connect-src 'self'; img-src 'self' data:; object-src 'none'; frame-ancestors 'none'")
        self.send_header('Content-Length', str(len(raw)))
        self.end_headers()
        try:
            self.wfile.write(raw)
        except (BrokenPipeError, ConnectionResetError):
            pass

    def _json(self, code: int, data: Any):
        self._reply(code, json.dumps(self.server.core.as_json(data), ensure_ascii=False, default=str).encode('utf-8'))

    def _auth(self) -> bool:
        header = self.headers.get('Authorization', '')
        return secrets.compare_digest(header, 'Bearer ' + self.server.api_token)

    def _payload(self):
        n = int(self.headers.get('Content-Length', '0'))
        limit=12_000_000 if urllib.parse.urlsplit(self.path).path=='/studio/run-file' else MAX_BODY
        if n < 1 or n > limit:
            raise ValueError('Payload exceeds limit')
        result = json.loads(self.rfile.read(n))
        if not isinstance(result, dict):
            raise ValueError('JSON object required')
        return result

    def _act(self, callback):
        try:
            callback()
        except KeyError as exc:
            self._json(404, {'error': str(exc)})
        except (ValueError, TypeError, PermissionError, json.JSONDecodeError) as exc:
            self._json(400, {'error': str(exc)})
        except Exception as exc:
            log.exception('Request failed')
            self._json(500, {'error': f'{type(exc).__name__}: {exc}'})

class ThreadedHTTPServer(http.server.ThreadingHTTPServer):
    daemon_threads = True
    allow_reuse_address = True


def make_server(host='127.0.0.1',port=8765,db=None,token=None):
    ip=__import__('ipaddress').ip_address(host)
    if not ip.is_loopback:
        raise ValueError('GUI can bind only to loopback; remote exposure is unsupported')
    core=import_core()
    server=ThreadedHTTPServer((host,port),StudioHandler)
    server.core=core
    server.audit=read_audit()
    server.api_token=token or secrets.token_urlsafe(32)
    base=Path(db).resolve() if db else ROOT/'traceatlas_studio_data'/'cases.sqlite3'
    base.parent.mkdir(parents=True,exist_ok=True)
    server.ledger=core.CaseLedger(base)
    return server


def self_test():
    audit=read_audit()
    result={'studio':'PASS','syntax_valid_sections':audit['syntax_ok'],'damaged_sections':audit['damaged_count'],'original_sha256_match':hashlib.sha256(original420k_bytes()).hexdigest()==audit['original_sha256'],'domain_test':None,'errors':[]}

        result['domain_test']=rep.get('status')
        result['domain_smoke']=rep.get('smoke')
        if rep.get('status')!='PASS':result['errors'].extend(rep.get('errors',[]))
    except Exception as exc:
        result['errors'].append(str(exc))
    if not result['original_sha256_match']:result['errors'].append('Original archive SHA-256 mismatch')
    result['status']='PASS' if not result['errors'] else 'FAIL'
    return result

    try:server.serve_forever(poll_interval=.3)
    except KeyboardInterrupt:pass
    finally:server.server_close()
    return 0
